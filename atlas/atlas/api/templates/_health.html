{# The one rule this macro exists to enforce: a health state ALWAYS carries its word, and UNKNOWN always carries the hatch. Colour never carries the meaning on its own — not for a colour-blind reader, not in a screenshot, not in a terminal rendering the HTML as text. Every surface that shows health goes through here. There is no second way to draw one. #} {# Anything we do not recognise is an absence of knowledge, not a new state. An unrecognised value previously passed straight through into the attribute, matched no CSS rule, and rendered with no colour, no dashed border and no hatch — which is much closer to OK than to UNKNOWN. #} {% macro normalise(health) -%} {% set h = (health or 'unknown')|lower|trim %} {{- h if h in ('ok', 'degraded', 'failing') else 'unknown' -}} {%- endmacro %} {% macro badge(health) -%} {% set h = normalise(health) %} {{ {'ok':'OK','degraded':'Degraded','failing':'Failing'}.get(h, 'Unknown') }} {%- endmacro %} {% macro rail(health) -%} {% set h = normalise(health) %} {%- endmacro %} {% macro severity(level) -%} {% set s = level|int %} {{ {0:'Info',1:'Normal',2:'High',3:'Critical'}.get(s, 'Normal') }} {%- endmacro %}
